Quick Start — Mass

One Mass call drives a full jurisdiction-aware DAG

typescript · @mass/primitives 
import { Mass } from '@mass/primitives';

const mass = new Mass({
  baseUrl: 'https://api.zone.mass.inc',
  token: process.env.MASS_API_TOKEN!,
});

await mass.operations.execute('entity.rename', entityId, {
  new_name: 'Acme Holdings Inc',
});

const note = await mass.instruments.create({
  instrument_class: 'convertible',
  jurisdiction_id: 'us-de',
  issuer_entity_id: entityId,
  metadata: { discount_rate: '20%', valuation_cap: '12000000' },
});
Quick Start — Moxie

One Moxie call settles an oracle event, pays winners, issues assets, nets margin

typescript · @moxie/sdk 
import { Moxie } from '@moxie/sdk';

const moxie = new Moxie({
  token: process.env.MOXIE_TOKEN!,
  region: 'us-east-1',
});

// 1. Pre-flight the 23-domain compliance tensor.
const preflight = await moxie.compliance.preflight({
  brand_id: 'bout',
  entity_ids: [entityId],
});

// 2. Atomic settlement: resolve event, pay winners, issue scarves, net margin.
const batch = await moxie.settlements.batch({
  idempotency_key: 'ufc-317-card-1',
  items: [
    { type: 'pay_winners', market_id, winners },
    { type: 'issue_asset', template_id, to: winners },
    { type: 'net_margin',  player_ids: winners },
  ],
  finality_target: 'finalized',
  compliance_commitment: preflight.commitment,
});

// One block. One receipt. One replay hash.
const receipt = await batch.receipt();
